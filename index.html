Check your email (including spam) for verification."); }); }) .catch((error) => { alert("❌ " + error.message); }); }); // LOGIN document.getElementById("login-btn").addEventListener("click", () => { const email = document.getElementById("email").value.trim(); const password = document.getElementById("password").value.trim(); if (!email || !password) { alert("❌ Please enter both email and password."); return; } auth.signInWithEmailAndPassword(email, password) .then((userCredential) => { const user = userCredential.user; if (!user.emailVerified) { alert("⚠️ Please verify your email before accessing Kairah Studio."); auth.signOut(); return; } alert("✅ Login Successful.
